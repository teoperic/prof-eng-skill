% \docneedspace{len}: start a new page unless len is left. Directly under a heading
% (\if@nobreak) a break must not fall between the two, so if room is short it overfills
% the page at a point where no break is allowed: TeX then breaks before the heading
% and the heading moves to the next page together with what follows (the same trick
% tcolorbox uses for its 'nobreak' handling). The overfill is len plus the page's
% shrink, so shrinkable glue cannot absorb it, yet small enough to fit on the next page;
% \@nobreak is set again because the empty paragraph of the trick clears it.
\makeatletter
\newdimen\doc@push
\newcommand{\docneedspace}[1]{%
  \par
  \if@nobreak
    \ifdim\dimexpr\pagegoal-\pagetotal\relax<#1\relax
      \doc@push=\dimexpr#1+\pageshrink+1pt\relax
      \vspace{\doc@push}{\parskip\z@\nointerlineskip\noindent}\vspace{-\doc@push}\par
      \global\@nobreaktrue
    \fi
  \else
    \Needspace{#1}%
  \fi}
\makeatother
% A heading moves to the next page unless this much room is left under it.

% \begin{recipecard}{R3 -- Step response}{label} ... \end{recipecard}; same for solutionbox
% profeng/needspace=<len> runs \docneedspace before the box; it adds to the code that
% 'before skip' sets ('before=' would replace it). Directly under a heading tcolorbox
% skips that code and moves the heading itself if the box cannot start on the page,
% but it also adds a \penalty0 there, which lets TeX break between heading and box:
% make that a \nobreak.

\patchcmd{\tcb@prepare@break@operation}{\addpenalty\z@}{\iftcb@nobreak\nobreak\else\addpenalty\z@\fi}{}
  {\typeout{prof-eng: could not patch tcolorbox, a heading may end a page above a box}}
\makeatother
